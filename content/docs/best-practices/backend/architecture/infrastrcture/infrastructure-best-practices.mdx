## Shared Column Utility Patterns

<BestPractice title="Shared Column Utility Patterns">

Use a shared columns utility for repeated technical columns (ids, timestamps, etc.) instead of redefining them in every table.

## !dont copy-paste-columns

**Repeat common columns manually in every table.** This leads to:

- Inconsistent column names and types across tables  
- Hard-to-apply global changes (e.g., changing timestamp mode)  
- Copy–paste errors when adding new tables  

```typescript !example
// ❌ Bad: Each table redefines common columns by hand

// File: infra/database/tables/user.table.ts
export const userTable = pgTable("users", {
  id: uuid("id").primaryKey().notNull(),
  createdAt: timestamp("createdAt", { mode: "date" }).notNull().defaultNow(),
  updatedAt: timestamp("updatedAt", { mode: "date" }).notNull().defaultNow(),
  email: varchar("email", { length: 320 }),
});

// File: infra/database/tables/document.table.ts
export const documentTable = pgTable("documents", {
  id: uuid("id").primaryKey().notNull(),                         // duplicated
  createdAt: timestamp("createdAt", { mode: "date" }).notNull().defaultNow(), // duplicated
  updatedAt: timestamp("updatedAt", { mode: "date" }).notNull().defaultNow(), // duplicated
  title: varchar("title", { length: 255 }),
});
````

## !do shared-columns-utility

**Extract shared columns into a tiny helper module.** This provides:

* **Consistent column definitions** for ids and timestamps
* **Single place** to change types/defaults for all tables
* **Clear intent** about “technical” columns vs business columns

```typescript !example
// File: infra/database/utils/shared-cols.ts
import { timestamp, uuid } from "drizzle-orm/pg-core";

// Small helper for UUID columns
export const UuidCol = (name: string) => uuid(name);

// Shared technical columns for all tables
export const SharedColumns = {
  id: UuidCol("id").primaryKey().notNull(),
  createdAt: timestamp("createdAt", { mode: "date" }).notNull().defaultNow(),
  updatedAt: timestamp("updatedAt", { mode: "date" }).notNull().defaultNow(),
};
```

```typescript !example
// File: infra/database/tables/user.table.ts
import { SharedColumns } from "@infra/database/utils/shared-cols";

export const userTable = pgTable("users", {
  ...SharedColumns, // id, createdAt, updatedAt
  email: varchar("email", { length: 320 }),
});

// File: infra/database/tables/document.table.ts
import { SharedColumns } from "@infra/database/utils/shared-cols";

export const documentTable = pgTable("documents", {
  ...SharedColumns, // id, createdAt, updatedAt
  title: varchar("title", { length: 255 }),
});
```

</BestPractice>

## Serialization & Flattening Patterns for nested objects

<BestPractice title="Serialization & Flattening Patterns for nested objects">

Use schema-driven serialization to flatten nested domain objects into DB-friendly rows and to reconstruct entities from rows.

## !dont ad-hoc-nested-mapping

**Manually map nested objects inline in each repository method.** This causes:

* Repeated, slightly different mapping logic in multiple places
* Easy-to-miss fields when the domain model changes
* Hidden coupling between DB shapes and domain shapes

```typescript !example
// ❌ Bad: Manual, scattered mapping logic in each repository method

const host = /* Host entity from domain */;

// Repository manually flattens nested objects into columns
const row = {
  id: host.id,
  firstName: host.firstName,
  lastName: host.lastName,

  // address flattened by hand
  country: host.address.country,
  city: host.address.city,
  state: host.address.state,
  zipCode: host.address.zipCode,
  address: host.address.address,

  // socialLinks flattened by hand
  websiteLink: host.socialLinks.websiteLink,
  instagramLink: host.socialLinks.instagramLink,
  // ...
};

// Another method repeats a similar manual mapping for a different query…
```

## !do schema-driven-serialization

**Use a single serialization / deserialization path (e.g. Effect Schema) to flatten and rebuild entities.** This provides:

* **Centralized mapping** between domain entity ↔ serialized shape ↔ DB row
* **Safer nested handling** for value objects and aggregates
* **Reusable helpers** in the repository instead of ad-hoc mapping

```typescript !example
// File: domain/user/user.schema.ts
import * as S from "@effect/schema/Schema";

export const UserSchema = S.Struct({
  id: S.String,          // already branded at the domain layer
  firstName: S.String,
  lastName: S.String,
  // address + socialLinks are already flattened to DB-friendly shape here
  country: S.String,
  city: S.String,
  websiteLink: S.optional(S.String),
});

export type SerializedUser = S.Schema.Encoded<typeof UserSchema>;
```

```typescript !example
// File: domain/user/user.entity.ts
export class User {
  // ...

  // Domain entity → SerializedUser (flattened)
  serialized(): E.Effect<SerializedUser, SerializationError, never> {
    return S.encode(UserSchema)(this).pipe(
      E.mapError(
        (err) =>
          new SerializationError("Failed to serialize User", "User", err),
      ),
    );
  }

  // SerializedUser → Domain entity
  static create(
    input: SerializedUser,
  ): E.Effect<User, DeserializationError, never> {
    return S.decodeUnknown(UserSchema)(input).pipe(
      E.map((data) => new User(data)),
      E.mapError(
        (err) =>
          new DeserializationError(
            "Failed to deserialize User",
            "User",
            err,
          ),
      ),
    );
  }
}
```

```typescript !example
// File: infra/repositories/user.drizzle.repository.ts

export class UserDrizzleRepository extends UserRepository {
  constructor(private readonly db: any) {
    super();
  }

  // SerializedUser -> DB row
  private toDbSerialized(serialized: SerializedUser,): E.Effect<SerializedUser, SerializationError, never> 
  {
    return E.succeed(serialized);
  }

  // DB row -> domain User entity using the domain create()
  private fromDbRow(row: any): E.Effect<User, QueryError, never> {
    return pipe(
      User.create(row as SerializedUser),
      E.mapError(
        (err) =>
          new DeserializationError(
            "Failed to deserialize User from DB",  "User", err,
        ),
      ),
    );
  }

  // Example: domain User -> SerializedUser -> insert into DB
  add(entity: User): RepositoryEffect<User, MutationError> {
    return pipe(
      entity.serialized(), // User -> SerializedUser (flattened)
      E.flatMap((serialized) => this.toDbSerialized(serialized)),
      E.flatMap((dbData) =>
        E.tryPromise({
          try: () => this.db.insert(users).values(dbData),
          catch: (error) =>
            new UserMutationError(
              "add",  `Failed to add user: ${error}`, "User", entity.id,
            ),
        }),
      ),
      E.as(entity),
    );
  }
}

```

## Query Helper Utility Patterns

<BestPractice title="Query Helper Utility Patterns">

Extract reusable query helpers (execute, map, single/multiple) instead of re-implementing query + error + mapping logic in every method.

## !dont repeated-query-logic

**Inline query execution and mapping in every repository method.** This causes:

* Duplicated error handling and logging
* Slightly different behaviors for “single vs multiple” queries
* Harder to introduce cross-cutting concerns (timeouts, metrics, etc.)

```typescript !example
// ❌ Bad: Each method does its own query + error + mapping

fetchByUserId(userId: UUID): RepositoryEffect<O.Option<Host>, HostNotFoundError> {
  return E.tryPromise({
    try: () =>
      this.db.select().from(hosts).where(eq(hosts.userId, userId)).limit(1),
    catch: () => new HostNotFoundError(),
  }).pipe(
    E.flatMap((rows) => {
      if (rows.length === 0) {
        return E.succeed(O.none());
      }

      // Inline DB row -> Host mapping
      const row = rows[0];
      const serializedHost = {
        ...row,
        address: { /* ... */ },
        socialLinks: { /* ... */ },
      };

      return Host.create(serializedHost).pipe(
        E.map(O.some),
        E.mapError(() => new HostNotFoundError()),
      );
    }),
  );
}
```

## !do reusable-query-helpers

**Centralize query execution and mapping in small helpers.** This provides:

* **Single place** to handle driver errors and wrap them in repository errors
* **Consistent behavior** for “single” and “multiple” fetch patterns
* **Cleaner repository methods** focused only on *what* they query

```typescript !example
// File: infra/database/utils/execute-query.ts
import { Effect as E } from "effect";

/**
 * Wrap a DB Promise in an Effect and map driver errors to a repository error.
 */
export function executeQuery<T, EErr>(
  query: () => Promise<T>,
  onError: (cause: unknown) => EErr,
): E.Effect<T, EErr> {
  return E.tryPromise({
    try: query,
    catch: onError,
  });
}
```

```typescript !example
// File: infra/repositories/host.drizzle.repository.ts
type HostRow = typeof hosts.$inferSelect;

const rowToHost = (row: HostRow): E.Effect<Host, HostNotFoundError> => {
  // Map flat DB row into the serialized shape expected by Host.create().
  const serializedHost = {
    ...row,
    address: {
      country: row.country,
      city: row.city,
      state: row.state,
      zipCode: row.zipCode,
      address: row.address,
    },
    socialLinks: {
      websiteLink: row.websiteLink,
      venmoLink: row.venmoLink,
      googleReviewLink: row.googleReviewLink,
      facebookLink: row.facebookLink,
      instagramLink: row.instagramLink,
      twitterLink: row.twitterLink,
    },
  };

  return pipe(
    Host.create(serializedHost),
    E.mapError(() => new HostNotFoundError()),
  );
};

export class HostDrizzleRepository {
  constructor(private readonly db: DrizzleDB) {}

  // One row or none
  private fetchSingle(
    query: () => Promise<HostRow[]>,
  ): E.Effect<O.Option<Host>, HostNotFoundError> {
    return pipe(
      executeQuery(query, () => new HostNotFoundError()),
      E.map(O.fromIterable),
      E.flatMap((option) =>
        O.match(option, {
          onNone: () => E.succeed(O.none()),
          onSome: (row) => pipe(rowToHost(row), E.map(O.some)),
        }),
      ),
    );
  }

  // Many rows
  private fetchMultiple(
    query: () => Promise<HostRow[]>,
  ): E.Effect<readonly Host[], HostNotFoundError> {
    return pipe(
      executeQuery(query, () => new HostNotFoundError()),
      E.flatMap((rows) => E.all(rows.map((row) => rowToHost(row)))),
    );
  }

  // Repository methods become small and declarative:

  fetchByUserId(
    userId: UUID,
  ): RepositoryEffect<O.Option<Host>, HostNotFoundError> {
    return this.fetchSingle(() =>
      this.db.select().from(hosts).where(eq(hosts.userId, userId)).limit(1),
    );
  }

  fetchByPhoneNumber(
    phoneNumber: number,
  ): RepositoryEffect<readonly Host[], HostNotFoundError> {
    return this.fetchMultiple(() =>
      this.db.select().from(hosts).where(eq(hosts.phoneNumber, phoneNumber)),
    );
  }
}
```

</BestPractice>
